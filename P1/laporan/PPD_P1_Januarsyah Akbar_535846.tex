\chapter[Dasar Teori]{Dasar Teori}

\section{NumPy (Numerical Python)}
\textit{NumPy} adalah pustaka inti dalam ekosistem komputasi ilmiah berbasis Python yang menyediakan objek array multidimensi berkinerja tinggi, dikenal sebagai \texttt{ndarray} (\textit{N-dimensional array}). Dibandingkan dengan tipe data bawaan \textit{list} pada Python murni, \textit{ndarray} menyimpan data dalam blok memori yang kontinu secara homogen (\textit{contiguous homogeneous memory layout}), memungkinkan operasi matematika vektorisasi (\textit{vectorization}) dieksekusi secara cepat tanpa beban komputasi perulangan eksplisit.

Fitur-fitur fundamental \textit{NumPy} meliputi:
\begin{itemize}
    \item \textbf{Array Creation \& Attributes}: Pembuatan array satu dimensi hingga multidimensi serta inspeksi bentuk dimensi melalui atribut \texttt{shape}, tipe data elemen \texttt{dtype}, dan ukuran data \texttt{size}.
    \item \textbf{Indexing dan Slicing}: Pengambilan elemen spesifik berdasarkan indeks berbasis nol (\textit{zero-based indexing}) serta pemotongan subset array menggunakan sintaks rentang \texttt{[start:stop:step]} tanpa perlu menyalin data fisik (\textit{memory view}).
    \item \textbf{Reshaping \& Flattening}: Pengubahan geometri array tanpa mengubah nilai elemen melalui metode \texttt{reshape()}, serta perataan ke 1D melalui \texttt{ravel()} atau \texttt{reshape(-1)}.
    \item \textbf{Array Operations}: Operasi aritmatika elemen-ke-elemen (\textit{element-wise operations}), penggabungan (\texttt{concatenate}), pencarian berbasis kondisi (\texttt{np.where}), dan fungsi statistik (\texttt{sum}, \texttt{mean}, \texttt{std}, \texttt{prod}).
\end{itemize}

\section{Pandas}
\textit{Pandas} adalah pustaka analisis dan manipulasi data terstruktur yang dibangun di atas \textit{NumPy}. \textit{Pandas} menyediakan dua struktur data utama:
\begin{enumerate}
    \item \textbf{Series}: Struktur data satu dimensi berlabel yang mampu menampung berbagai tipe data (integer, string, float, dsb.).
    \item \textbf{DataFrame}: Struktur data dua dimensi berbentuk tabel dua sumbu (\textit{rows} dan \textit{columns}) dengan indeks fleksibel.
\end{enumerate}

Operasi krusial pada \textit{Pandas} meliputi:
\begin{itemize}
    \item \textbf{Data Cleaning \& Transformation}: Penambahan atau penghapusan kolom (\texttt{drop}), pengubahan tipe data, penanganan nilai hilang (\textit{missing values}), dan pemfilteran berbasis kondisi boolean.
    \item \textbf{Grouping \& Aggregation}: Operasi \textit{split-apply-combine} menggunakan metode \texttt{groupby()} yang dikombinasikan dengan \texttt{agg()} untuk menghitung berbagai metrik ringkasan (misalnya \textit{mean}, \textit{max}, \textit{min}, \textit{std}, \textit{count}) pada kelompok data tertentu.
    \item \textbf{Relational Operations (Merge, Join, Concat)}: Penggabungan beberapa tabel berdasarkan kolom kunci tertentu (\texttt{merge}) atau penggabungan berorientasi sumbu horizontal/vertikal (\texttt{concat}).
    \item \textbf{Reshaping}: Transformasi bentuk tabel melalui pembuatan tabel silang (\texttt{pivot\_table}) dan restrukturisasi dari format lebar (\textit{wide}) ke panjang (\textit{long}) menggunakan metode \texttt{melt}.
    \item \textbf{Functional Application}: Penerapan ekspresi logika kustom pada seluruh baris atau kolom melalui fungsi \texttt{apply()} dengan ekspresi \texttt{lambda}.
\end{itemize}

\chapter[Tugas dan Analisis]{Tugas dan Analisis}

\section{Soal 1: NumPy Reshape dan Boolean Filtering}
\subsection{Deskripsi Tugas}
Buatlah array NumPy 2D berukuran $4 \times 4$ berisi angka acak (rentang nilai 0 hingga 50), kemudian ubah bentuk dimensinya (\textit{reshape}) menjadi array $2 \times 8$, lalu tampilkan hanya elemen-elemen yang bernilai lebih dari 25.

\subsection{Implementasi Kode}
\begin{lstlisting}[style=pythonstyle, caption=Solusi Soal 1 - NumPy Reshape dan Filtering]
import numpy as np

# 1. Inisialisasi seed dan buat array 2D 4x4 bilangan bulat acak [0, 50]
np.random.seed(42)
arr_awal = np.random.randint(0, 51, size=(4, 4))
print("Array Asal (4x4):")
print(arr_awal)
print("Shape Asal:", arr_awal.shape)
print("-" * 40)

# 2. Reshape menjadi array berukuran 2x8
arr_reshaped = arr_awal.reshape(2, 8)
print("Hasil Reshape (2x8):")
print(arr_reshaped)
print("Shape Reshape:", arr_reshaped.shape)
print("-" * 40)

# 3. Filtering elemen bernilai > 25
hasil_filter = arr_reshaped[arr_reshaped > 25]
print("Elemen dengan nilai > 25:")
print(hasil_filter)
print("Banyaknya elemen yang lolos filter:", len(hasil_filter))
\end{lstlisting}

\subsection{Analisis dan Pembahasan}
\begin{enumerate}
    \item \textbf{Pembangkitan Angka Acak}: Fungsi \texttt{np.random.randint(0, 51, size=(4, 4))} menghasilkan matriks berukuran $4 \times 4$ dengan 16 bilangan bulat acak pada interval inklusif $[0, 50]$. Penggunaan \texttt{np.random.seed(42)} memastikan hasil komputasi bersifat deterministik dan dapat direplikasi secara konsisten.
    \item \textbf{Mekanisme Reshape}: Operasi \texttt{reshape(2, 8)} mengubah struktur matriks dari 4 baris $\times$ 4 kolom menjadi 2 baris $\times$ 8 kolom. Operasi ini berjalan dengan kompleksitas $O(1)$ karena tidak melakukan penyalinan data fisik di memori, melainkan hanya memodifikasi metadata \textit{stride} dan dimensi dari \textit{ndarray} yang tetap berjumlah $4 \times 4 = 2 \times 8 = 16$ elemen.
    \item \textbf{Boolean Masking}: Ekspresi \texttt{arr\_reshaped > 25} menghasilkan array boolean dengan dimensi yang sama dengan array asal. Ketika dimasukkan sebagai indeks (\texttt{arr\_reshaped[...]}), \textit{NumPy} melakukan ekstraksi elemen bernilai \texttt{True} dan mengembalikannya sebagai array satu dimensi (vektor).

\end{enumerate}

\section{Soal 2: Pandas Groupby dan Multiple Aggregations}
\subsection{Deskripsi Tugas}
Buatlah \textit{DataFrame} untuk data karyawan yang berisi kolom \texttt{Nama}, \texttt{Pekerjaan}, dan \texttt{Gaji}. Lalu lakukan operasi \texttt{groupby()} pada kolom \texttt{Pekerjaan} dan hitung nilai \textbf{maksimum}, \textbf{minimum}, serta \textbf{standar deviasi} untuk kolom \texttt{Gaji} menggunakan metode \texttt{agg()}.

\subsection{Implementasi Kode}
\begin{lstlisting}[style=pythonstyle, caption=Solusi Soal 2 - Pandas Groupby dan Aggregation]
import pandas as pd

# 1. Pembuatan DataFrame Karyawan
data_karyawan = {
    'Nama': ['Andi', 'Budi', 'Citra', 'Dewi', 'Eko', 'Fani', 'Gita', 'Hadi'],
    'Pekerjaan': ['Programmer', 'Programmer', 'Data Analyst', 'Data Analyst',
                  'Manager', 'Manager', 'Programmer', 'Data Analyst'],
    'Gaji': [6000000, 6500000, 5500000, 5800000, 9500000, 10000000, 7000000, 6200000]
}
df_karyawan = pd.DataFrame(data_karyawan)

# 2. Operasi Groupby dan Perhitungan Agregasi Berganda
ringkasan_gaji = df_karyawan.groupby('Pekerjaan')['Gaji'].agg(
    Gaji_Maksimum='max',
    Gaji_Minimum='min',
    Standar_Deviasi='std'
).reset_index()

print("Hasil Ringkasan Agregasi Gaji per Pekerjaan:")
print(ringkasan_gaji)
\end{lstlisting}

\subsection{Analisis dan Pembahasan}
\begin{enumerate}
    \item \textbf{Pengelompokan Kategori (\texttt{groupby})}: Metode \texttt{groupby('Pekerjaan')} mempartisi data menjadi kelompok-kelompok diskrit berdasarkan nama jabatan (\textit{Data Analyst}, \textit{Manager}, dan \textit{Programmer}).
    \item \textbf{Agregasi Terarah (\texttt{agg})}: Pemanggilan \texttt{.agg(Gaji\_Maksimum='max', Gaji\_Minimum='min', Standar\_Deviasi='std')} memungkinkan penghitungan beberapa metrik statistik deskriptif sekaligus dalam satu lintasan komputasi.
    \item \textbf{Interpretasi Metrik}:
    \begin{itemize}
        \item Nilai \texttt{max} dan \texttt{min} memperlihatkan batas rentang kompensasi tertinggi dan terendah pada masing-masing posisi pekerjaan.
        \item Nilai \texttt{std} (\textit{standard deviation}) menggambarkan dispersi atau variasi sebaran gaji di antara karyawan dalam jabatan yang sama.
    \end{itemize}
    \item \textbf{Normalisasi Indeks (\texttt{reset\_index})}: Mengembalikan kolom pengelompokan (\texttt{Pekerjaan}) menjadi kolom reguler tabel dua dimensi, bukan sebagai indeks baris berjenjang.
\end{enumerate}

\section{Soal 3: Pandas Relational Merge (Left Join dengan Kolom Kunci Berbeda dan Suffixes)}
\subsection{Deskripsi Tugas}
Buat dua \textit{DataFrame}: DataFrame karyawan dengan kolom \texttt{Emp\_id}, \texttt{Nama}, dan \texttt{Divisi}, serta DataFrame gaji dengan kolom \texttt{ID}, \texttt{Gaji}, dan \texttt{Status} (masing-masing minimal tiga baris). Setelah itu, gabungkan kedua \textit{DataFrame} tersebut menggunakan \texttt{merge()} dengan parameter \texttt{left\_on='Emp\_id'} dan \texttt{right\_on='ID'}, terapkan metode \textbf{left join}, dan gunakan parameter \texttt{suffixes} untuk membedakan nama kolom yang bertabrakan.

\subsection{Implementasi Kode}
\begin{lstlisting}[style=pythonstyle, caption=Solusi Soal 3 - Pandas Merge dengan Key Berbeda dan Suffixes]
import pandas as pd

# 1. DataFrame Karyawan (Tabel Kiri)
df_info = pd.DataFrame({
    'Emp_id': ['EMP01', 'EMP02', 'EMP03', 'EMP04'],
    'Nama': ['Rian', 'Sari', 'Doni', 'Lestari'],
    'Divisi': ['Engineering', 'Data Science', 'Product', 'Marketing'],
    'Status': ['Aktif', 'Aktif', 'Cuti', 'Aktif'] # Kolom yang bertabrakan
})

# 2. DataFrame Gaji (Tabel Kanan)
df_payroll = pd.DataFrame({
    'ID': ['EMP01', 'EMP02', 'EMP03', 'EMP05'], # EMP04 tidak ada di kanan, EMP05 tidak ada di kiri
    'Gaji': [7500000, 8200000, 6800000, 9000000],
    'Status': ['Tetap', 'Kontrak', 'Tetap', 'Kontrak'] # Kolom yang bertabrakan
})

# 3. Penggabungan Tabel Relasional
df_hasil_merge = pd.merge(
    df_info,
    df_payroll,
    left_on='Emp_id',
    right_on='ID',
    how='left',
    suffixes=('_karyawan', '_kepegawaian')
)

print("Hasil Penggabungan Data (Left Join):")
print(df_hasil_merge)
\end{lstlisting}

\subsection{Analisis dan Pembahasan}
\begin{enumerate}
    \item \textbf{Mekanisme Left Join (\texttt{how='left'})}: Menjamin bahwa seluruh 4 baris data dari DataFrame kiri (\texttt{df\_info}) dipertahankan secara utuh pada tabel luaran. Karyawan dengan ID \texttt{EMP04} (Lestari) yang tidak tercatat dalam tabel gaji kanan tetap muncul dalam luaran dengan nilai kolom \texttt{ID}, \texttt{Gaji}, dan \texttt{Status\_kepegawaian} otomatis terisi nilai \texttt{NaN} (\textit{Not a Number}).
    \item \textbf{Penanganan Kunci Berbeda (\texttt{left\_on} dan \texttt{right\_on})}: Ketika dua sumber data menggunakan nama atribut pengenal yang tidak identik (\texttt{Emp\_id} pada tabel kiri dan \texttt{ID} pada tabel kanan), spesifikasi eksplisit \texttt{left\_on} dan \texttt{right\_on} menjembatani proses pencocokan nilai entitas secara akurat.
    \item \textbf{Resolusi Kolisi Nama (\texttt{suffixes})}: Kedua tabel memiliki kolom bernama sama yaitu \texttt{Status} dengan arti kontekstual berbeda (status kehadiran aktif/cuti vs status kepegawaian tetap/kontrak). Parameter \texttt{suffixes=('\_karyawan', '\_kepegawaian')} secara otomatis mengubah nama kolom menjadi \texttt{Status\_karyawan} dan \texttt{Status\_kepegawaian}, mencegah penimpaan data dan menjaga integritas informasi.
\end{enumerate}

\chapter[Kesimpulan]{Kesimpulan}
Berdasarkan serangkaian langkah praktikum dan penyelesaian tugas pada Pertemuan I, dapat ditarik beberapa kesimpulan penting:
\begin{enumerate}
    \item \textbf{Efisiensi NumPy}: Pustaka \textit{NumPy} dengan objek \textit{ndarray} menyediakan kapabilitas komputasi matriks berkinerja tinggi, manipulasi bentuk array melalui \textit{reshaping} yang hemat memori, serta kemampuan \textit{vectorized boolean filtering} yang sangat efisien untuk seleksi data numerik bervolume besar.
    \item \textbf{Fleksibilitas Pandas}: Pustaka \textit{Pandas} mempermudah manipulasi data tabular melalui antarmuka \textit{DataFrame}, memudahkan pengelolaan indeks, pengurutan, pengelompokan data multidimensi melalui \texttt{groupby()} dan \texttt{agg()}, serta restrukturisasi tabel melalui operasi \textit{pivot} dan \textit{melt}.
    \item \textbf{Integritas Penggabungan Data}: Operasi \texttt{merge()} pada \textit{Pandas} menyediakan fleksibilitas relasional yang setara dengan operasi SQL \textit{JOIN}, di mana penggunaan parameter \texttt{left\_on}, \texttt{right\_on}, dan \texttt{suffixes} terbukti andal dalam menyatukan dataset heterogen tanpa kehilangan konteks maupun duplikasi kolom.
\end{enumerate}
